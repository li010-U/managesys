  if (s === 'running') return 'success'

function deviceStatusLabel(s: string): string {

const slotHeight = 36
const uSlots = computed(() => {
  const rack = viewRack.value
  if (!rack) return []
  const totalU = rack.total_units || 42
  const devices = rackDevices.value || []
  const slots: any[] = []
  for (let u = totalU; u >= 1; u--) {
    const dev = devices.find(d => d.start_u !== undefined && d.end_u !== undefined && u >= d.start_u && u <= d.end_u)
    if (dev) {
      const isFirst = u === dev.start_u
      const height = dev.end_u! - dev.start_u! + 1
      slots.push({ number: u, displayNumber: isFirst ? String(u) : '', occupied: true, device: dev, merged: !isFirst, firstOfMulti: isFirst, deviceHeight: height })
    } else {
      const isMerged = devices.some(d => d.start_u !== undefined && d.end_u !== undefined && u > d.start_u && u <= d.end_u && u !== d.start_u)
      slots.push({ number: u, displayNumber: String(u), occupied: false, device: null, merged: isMerged, firstOfMulti: true, deviceHeight: 1 })
    }
  }
  return slots
})

function getSlotStyle(slot: any) {
  if (slot.occupied && slot.firstOfMulti && slot.deviceHeight > 1) {
    return { minHeight: (slotHeight * slot.deviceHeight + (slot.deviceHeight - 1)) + 'px' }
  }
  if (slot.merged) return { display: 'none' }
  return { minHeight: slotHeight + 'px' }
}

const avgUtilization = computed(() => {
  if (rackList.value.length === 0) return 0
  const sum = rackList.value.reduce((a, r) => a + (r.total_units - r.available_units) / r.total_units * 100, 0)
  return Math.round(sum / rackList.value.length)
})
const avgUtilColor = computed(() => { const v = avgUtilization.value; return v >= 80 ? '#e74c3c' : v >= 50 ? '#e67e22' : '#27ae60' })

function rackUtilPercent(rack: RackInfo): number {
  if (!rack.total_units) return 0
  return Math.round((rack.total_units - rack.available_units) / rack.total_units * 100)
}
function rackUtilColor(rack: RackInfo): string { const u = rackUtilPercent(rack); return u >= 80 ? '#e74c3c' : u >= 50 ? '#e67e22' : '#27ae60' }
function utilClass(rack: RackInfo): string { const u = rackUtilPercent(rack); return u >= 80 ? 'rack-util-high' : u >= 50 ? 'rack-util-mid' : 'rack-util-low' }

function rackMatches(rack: RackInfo): boolean {
  if (!rackSearch.value) return true
  const q = rackSearch.value.toLowerCase()
  return rack.code.toLowerCase().includes(q) || rack.name.toLowerCase().includes(q)
}

  return rack.code + ' - ' + rack.name + '\\nU浣? ' + used + '/' + rack.total_units + ' (' + pct + '%)\\n璁惧: ' + (rack.device_count || 0) + '鍙癨\n鍔熺巼: ' + (rack.rated_power || '-') + 'kW'
}

async function fetchRooms() {
  try {
    const r = await getRoomsApi({ page: 1, page_size: 100 })
    roomList.value = r.data.items
    if (roomList.value.length > 0) {
      // Check if room_id query param is provided
      const qRoomId = Number(route.query.room_id)
      if (qRoomId && roomList.value.some(rm => rm.id === qRoomId)) {
        selectedRoomId.value = qRoomId
      } else {
        selectedRoomId.value = roomList.value[0].id
      }
      showRoomSelector.value = roomList.value.length > 1
      await fetchRacks()
    }
  } catch (e: any) {
    console.error('Failed to fetch rooms:', e)
  }
}
async function onRoomChange() { await fetchRacks() }
async function fetchRacks() {
  if (!selectedRoomId.value) return
  loading.value = true
  try { const r = await getRacksApi({ page: 1, page_size: 100, room_id: selectedRoomId.value }); rackList.value = r.data.items }
  catch { rackList.value = [] }
  finally { loading.value = false }
}

  dialog.value = { visible: true, isEdit: !!rack, id: rack?.id || 0 }
  if (rack) { form.value = { code: rack.code, name: rack.name, total_units: rack.total_units, rated_power: rack.rated_power?.toString() || undefined, row_pos: rack.row_pos, col_pos: rack.col_pos, description: rack.description || '' } }
  else { form.value = { code: '', name: '', total_units: 42, rated_power: undefined, row_pos: undefined, col_pos: undefined, description: '' } }
  setTimeout(() => formRef.value?.clearValidate(), 0)
}

  setTimeout(() => formRef.value?.clearValidate(), 0)
}
async function submit() {
  const valid = await formRef.value?.validate().catch(() => false)
  if (!valid) return
  if (!selectedRoomId.value) { ElMessage.warning('璇峰厛閫夋嫨鏈烘埧'); return }
  submitting.value = true
  try {
    const raw = { ...form.value, room_id: selectedRoomId.value }
    const numFields = ['row_pos', 'col_pos', 'rated_power', 'total_units']
    const data: any = {}
    for (const [k, v] of Object.entries(raw)) {
      if (v === '' || v === undefined) data[k] = null
      else if (numFields.includes(k)) data[k] = Number(v)
      else data[k] = v
    }
    if (dialog.value.isEdit) { await updateRackApi(dialog.value.id, data); ElMessage.success('宸叉洿鏂?) }
    else { await createRackApi(data); ElMessage.success('宸插垱寤?) }
    dialog.value.visible = false; await fetchRacks()
  } catch (e: any) {
    ElMessage.error(e?.response?.data?.detail || e?.message || '鎻愪氦澶辫触锛岃妫€鏌ュ悗绔湇鍔℃槸鍚﹁繍琛?)
  } finally { submitting.value = false }
}
